94    #[wasm_bindgen(constructor)]
95    pub fn new(
96        api_endpoint: web_sys::Url,
97        download_endpoint: web_sys::Url,
98    ) -> Result<WebUploaderClient, js_sys::Error> {
99        let api_endpoint =
100            url::Url::parse(&ToString::to_string(&api_endpoint.to_string())).unwrap();
101        let download_endpoint =
102            url::Url::parse(&ToString::to_string(&download_endpoint.to_string())).unwrap();
103        Ok(Self {
104            client: Rc::new(UploaderClient::new(
105                api_endpoint,
106                download_endpoint,
107                Transport::Both,
108            )),
109        })
110    }
111
112    #[wasm_bindgen(js_name = provisionFile)]
113    pub async fn provision_file(
114        &self,
115        file: WebUploadableFile,
116        file_name: String,
117    ) -> Result<WebProvisionedFile, js_sys::Error> {
118        let file_state = UploadFile { file, offset: 0, pending_read: Default::default() };
119        let provisioned_file = self
120            .client
121            .provision_file_async(file_state, file_name)
122            .await?;
123        Ok(WebProvisionedFile { inner: Some(provisioned_file) })
124    }
125
126    #[wasm_bindgen(js_name = uploadFile)]
127    pub async fn upload_file(
128        self,
129        provisioned_file: &mut WebProvisionedFile,
130        event_handler: Option<WebUploadEventHandler>,
131    ) -> Result<(), js_sys::Error> {
132        let provisioned_file = provisioned_file
133            .inner
134            .take()
135            .expect("ProvisionedFile used after being consumed");
136        let upload_progress = self.client.upload_provisioned_file_async(provisioned_file);
137        pin_mut!(upload_progress);
138        while let Some(progress_state) = upload_progress.try_next().await? {
139            if let Some(event_handler) = &event_handler {
140                event_handler.upload_progress(progress_state.current, progress_state.total)?;
141            }
142        }
143        Ok(())
144    }
145}

168impl AsyncRead for UploadFile {
169    fn poll_read(
170        mut self: Pin<&mut Self>,
171        cx: &mut task::Context<'_>,
172        buf: &mut [u8],
173    ) -> Poll<io::Result<usize>> {
174        let data = match mem::take(&mut self.pending_read) {
175            pending_read @ (UploadFileReadState::Reading { .. } | UploadFileReadState::Idle) => {
176                self.pending_read = pending_read;
177                let pending_read = match &mut self.pending_read {
178                    UploadFileReadState::Reading { pending_read } => pending_read,
179                    UploadFileReadState::Idle => {
180                        let file: WebUploadableFile = self.file.clone().into();
181                        let offset = self.offset;
182                        let len = buf.len().try_into().unwrap();
183                        let read_future = Box::pin(async move { file.read_at(offset, len).await });
184                        self.pending_read.insert(read_future)
185                    }
186                    UploadFileReadState::Available { .. } => unreachable!(),
187                };
188                let data = ready!(pending_read.poll_unpin(cx))
189                    .map_err(UploadFileReadError::from)?
190                    .dyn_into::<Uint8Array>()
191                    .expect("UploadableFile.read_at returns a Uint8Array");
192                self.pending_read = UploadFileReadState::Idle;
193                self.offset += u64::from(data.length());
194                data
195            }
196            UploadFileReadState::Available { data } => data,
197        };
198
199        let data_len = usize::try_from(data.length()).expect("usize is at least 32 bits");
200        let read_len = data_len.min(buf.len());
201        debug!("returning {read_len} of {data_len} bytes");
202        data.subarray(0, read_len as u32)
203            .copy_to(&mut buf[..read_len]);
204        if data_len > buf.len() {
205            self.pending_read = UploadFileReadState::Available {
206                data: data.subarray(buf.len() as u32, data.length()),
207            };
208        }
209        Poll::Ready(Ok(read_len))
210    }
211}

213impl UploadFileReadState {
214    fn insert(&mut self, pending_read: PendingRead) -> &mut PendingRead {
215        *self = Self::Reading { pending_read };
216        match self {
217            Self::Reading { pending_read } => pending_read,
218            _ => unreachable!(),
219        }
220    }
221}

223impl From<js_sys::Error> for UploadFileReadError {
224    fn from(from: js_sys::Error) -> Self {
225        let message = from
226            .dyn_ref::<JsString>()
227            .map(|string| format!("{string}"))
228            .unwrap_or_else(|| format!("{from:?}"));
229        Self { message }
230    }
231}
