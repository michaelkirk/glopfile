18    async fn derive_new(base_key: &CipherKey, hkdf_info: &[u8]) -> Self
19    where
20        Self: Sized,
21    {
22        let window = window().expect("web Window context present");
23        let crypto = window.crypto().expect("Window.crypto accessible");
24        let subtle = crypto.subtle();
25
26        let base_key_js = {
27            let key_usages = ["deriveKey"]
28                .into_iter()
29                .map(JsString::from)
30                .collect::<Array>();
31
32            let key_result = subtle
33                .import_key_with_str(
34                    "raw",
35                    &Uint8Array::from(&base_key.bytes()[..]),
36                    "HKDF",
37                    false,
38                    &key_usages,
39                )
40                .expect("provided valid parameters to SubtleCrypto.importKey");
41
42            JsFuture::from(key_result)
43                .await
44                .expect("SubtleCrypto.importKey promise returns a value")
45                .dyn_into::<CryptoKey>()
46                .expect("SubtleCrypto.importKey promise returns a CryptoKey")
47        };
48
49        let derived_key = {
50            let salt = Uint8Array::new_with_length(0);
51            let algorithm = HkdfParams::new(
52                "HKDF",
53                &JsString::from("SHA-256"),
54                &Uint8Array::from(hkdf_info),
55                &salt,
56            );
57
58            let key_usages = ["encrypt", "decrypt"]
59                .into_iter()
60                .map(JsString::from)
61                .collect::<Array>();
62
63            let key_size_bits = 8 * KEY_SIZE;
64            let derived_key_type = AesKeyGenParams::new(
65                "AES-GCM",
66                key_size_bits.try_into().expect("invalid key size"),
67            );
68
69            let key_result = subtle
70                .derive_key_with_object_and_object(
71                    &algorithm,
72                    &base_key_js,
73                    &derived_key_type,
74                    false,
75                    &key_usages,
76                )
77                .expect("provided valid parameters to SubtleCrypto.deriveKey");
78
79            JsFuture::from(key_result)
80                .await
81                .expect("SubtleCrypto.deriveKey promise returns a value")
82                .dyn_into::<CryptoKey>()
83                .expect("SubtleCrypto.deriveKey promise returns a CryptoKey")
84        };
85
86        Self { subtle, key: derived_key }
87    }
88
89    async fn encrypt(&self, mut plaintext_and_nonce: ContentCipherBuffer, aad: Vec<u8>) -> Vec<u8> {
90        let ContentCipherBufferPartsMut { nonce, mut data } = plaintext_and_nonce.parts_mut();
91        let plaintext = data.plaintext_mut();
92
93        let nonce = Uint8Array::from(&nonce[..]);
94        let aad = Uint8Array::from(&aad[..]);
95
96        let params = AesGcmParams::new("AES-GCM", &nonce);
97        params.set_additional_data(&aad);
98        let encryption_promise = self
99            .subtle
100            .encrypt_with_object_and_u8_array(&params, &self.key, plaintext)
101            .expect("provided valid parameters to SubtleCrypto.encrypt");
102        let output = JsFuture::from(encryption_promise)
103            .await
104            .expect("SubtleCrypto.encrypt promise returns a value");
105
106        let ciphertext = data.ciphertext_mut();
107        Uint8Array::new(&output).copy_to(ciphertext);
108        plaintext_and_nonce.into_nonce_and_ciphertext()
109    }
110
111    async fn decrypt(
112        &self,
113        mut ciphertext_and_nonce: ContentCipherBuffer,
114        aad: Vec<u8>,
115    ) -> Result<Bytes> {
116        let ContentCipherBufferPartsMut { nonce, mut data } = ciphertext_and_nonce.parts_mut();
117        let ciphertext = data.ciphertext_mut();
118
119        let nonce = Uint8Array::from(&nonce[..]);
120        let aad = Uint8Array::from(&aad[..]);
121
122        let params = AesGcmParams::new("AES-GCM", &nonce);
123        params.set_additional_data(&aad);
124        let encryption_promise = self
125            .subtle
126            .decrypt_with_object_and_u8_array(&params, &self.key, ciphertext)
127            .expect("provided valid parameters to SubtleCrypto.encrypt");
128        let output = JsFuture::from(encryption_promise)
129            .await
130            .map_err(|_| Error::Decrypt)?;
131
132        let plaintext = data.plaintext_mut();
133        Uint8Array::new(&output).copy_to(plaintext);
134        Ok(ciphertext_and_nonce.into_plaintext())
135    }
