1use std::path::Path;
2
3use uniffi_bindgen::bindings::{generate, GenerateOptions, TargetLanguage};
4
5pub use uniffi_bindgen::bindings;
6
7/// The glopfile UDL, resolved at compile time so callers don't need the source tree.
8const GLOPFILE_UDL: &str = concat!(env!("CARGO_MANIFEST_DIR"), "/../glopfile/src/lib.udl");
9
10pub fn write_bindings(
11    config_path: Option<impl AsRef<Path>>,
12    out_dir: impl AsRef<Path>,
13    language: TargetLanguage,
14    try_format_code: bool,
15) -> anyhow::Result<()> {
16    generate(GenerateOptions {
17        languages: vec![language],
18        source: GLOPFILE_UDL.into(),
19        out_dir: path_arg(out_dir.as_ref())?,
20        config_override: config_path
21            .map(|path| path_arg(path.as_ref()))
22            .transpose()?,
23        format: try_format_code,
24        ..GenerateOptions::default()
25    })
26}
27
28fn path_arg(path: &Path) -> anyhow::Result<camino::Utf8PathBuf> {
29    camino::Utf8PathBuf::from_path_buf(path.to_path_buf())
30        .map_err(|path| anyhow::anyhow!("path is not valid utf-8: {}", path.display()))
31}
