15pub struct NativeCipher {
16    cipher: Arc<Aes256Gcm>,
17}

19#[async_trait::async_trait(?Send)]
20impl super::CipherImpl for NativeCipher {
21    async fn derive_new(base_key: &CipherKey, hkdf_info: &[u8]) -> Self {
22        let mut aes_key = [0; KEY_SIZE];
23        let hkdf = Hkdf::<Sha256>::new(None, base_key.bytes());
24        hkdf.expand(hkdf_info, &mut aes_key)
25            .expect("failed to derive key");
26
27        Self { cipher: Arc::new(Aes256Gcm::new(&aes_key.into())) }
28    }
29
30    async fn encrypt(&self, mut plaintext_and_nonce: ContentCipherBuffer, aad: Vec<u8>) -> Vec<u8> {
31        // TODO: handle invalid crypt
32        let cipher = Arc::clone(&self.cipher);
33        spawn_blocking(move || {
34            let ContentCipherBufferPartsMut { nonce, mut data } = plaintext_and_nonce.parts_mut();
35            let payload = data.plaintext_mut();
36
37            let new_tag = cipher
38                .encrypt_inout_detached((&*nonce).into(), &aad, payload.into())
39                .expect("encryption failure");
40            *data.ciphertext_parts_mut().tag = new_tag.into();
41            plaintext_and_nonce.into_nonce_and_ciphertext()
42        })
43        .await
44        .unwrap_or_else(|panic| resume_unwind(panic.into_panic()))
45    }
46
47    async fn decrypt(
48        &self,
49        mut ciphertext_and_nonce: ContentCipherBuffer,
50        aad: Vec<u8>,
51    ) -> Result<Bytes> {
52        let cipher = Arc::clone(&self.cipher);
53        spawn_blocking(move || {
54            let ContentCipherBufferPartsMut { nonce, mut data } = ciphertext_and_nonce.parts_mut();
55            let ContentCipherBufferCiphertextPartsMut { payload, tag } =
56                data.ciphertext_parts_mut();
57            cipher
58                .decrypt_inout_detached((&*nonce).into(), &aad, payload.into(), (&*tag).into())
59                .map_err(|_| Error::Decrypt)?;
60            Ok(ciphertext_and_nonce.into_plaintext())
61        })
62        .await
63        .unwrap_or_else(|panic| resume_unwind(panic.into_panic()))
64    }
65}
