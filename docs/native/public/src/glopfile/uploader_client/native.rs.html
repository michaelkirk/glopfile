19impl UploaderClient {
20    pub fn provision_file(&self, path: &std::path::Path) -> Result<NativeProvisionedFile> {
21        current_thread_block_on(async {
22            let file = tokio::fs::File::open(path).await?;
23            let file_name = path
24                .file_name()
25                .ok_or(Error::InvalidInput("invalid file path"))?
26                .to_string_lossy()
27                .to_string();
28
29            let file = NativeUploadFile { file: file.compat() };
30            self.provision_file_async(file, file_name).await
31        })
32    }
33
34    pub fn upload_provisioned_file<F: FnMut(ProgressState<u64>)>(
35        &self,
36        provisioned_file: NativeProvisionedFile,
37        mut progress_fun: F,
38    ) -> Result<()> {
39        current_thread_block_on(async {
40            let upload_progress = self.upload_provisioned_file_async(provisioned_file);
41            pin_mut!(upload_progress);
42            while let Some(progress_state) = upload_progress.try_next().await? {
43                progress_fun(progress_state);
44            }
45            Ok(())
46        })
47    }
48}
